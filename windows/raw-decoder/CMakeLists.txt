cmake_minimum_required(VERSION 3.20)
project(ZFilterRawDecoder LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

set(LIBRAW_SOURCE "" CACHE PATH "Path to the pinned LibRaw source checkout")
if(NOT EXISTS "${LIBRAW_SOURCE}/libraw/libraw.h")
    message(FATAL_ERROR "LIBRAW_SOURCE must point to the pinned Nikon HE LibRaw checkout")
endif()
file(GLOB_RECURSE raw_sources CONFIGURE_DEPENDS "${LIBRAW_SOURCE}/src/*.cpp")
list(FILTER raw_sources EXCLUDE REGEX "_ph[.]cpp$")
add_library(zfilter_libraw STATIC ${raw_sources})
target_include_directories(zfilter_libraw PUBLIC "${LIBRAW_SOURCE}")
target_compile_definitions(zfilter_libraw PUBLIC LIBRAW_NODLL)
if(MSVC)
    target_compile_definitions(zfilter_libraw PUBLIC WIN32 NOMINMAX)
endif()

add_executable(zfilter-raw decode.cpp)
target_link_libraries(zfilter-raw PRIVATE zfilter_libraw)
if(MSVC)
    target_link_options(zfilter-raw PRIVATE /STACK:8388608)
endif()
