cmake_minimum_required(VERSION 3.20)
project(ZFilterRawDecoder LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(LIBRAW_SOURCE "" CACHE PATH "Path to the pinned LibRaw source checkout")
if(NOT EXISTS "${LIBRAW_SOURCE}/samples/dcraw_emu.cpp")
    message(FATAL_ERROR "LIBRAW_SOURCE must point to the pinned Nikon HE LibRaw checkout")
endif()
file(GLOB_RECURSE raw_sources CONFIGURE_DEPENDS "${LIBRAW_SOURCE}/src/*.cpp")
list(FILTER raw_sources EXCLUDE REGEX "_ph[.]cpp$")
add_library(zfilter_libraw SHARED ${raw_sources})
target_include_directories(zfilter_libraw PUBLIC "${LIBRAW_SOURCE}")
target_compile_definitions(zfilter_libraw PRIVATE LIBRAW_BUILDLIB)
set_target_properties(zfilter_libraw PROPERTIES OUTPUT_NAME "zfilter-libraw")
if(MSVC)
    target_compile_definitions(zfilter_libraw PRIVATE WIN32 NOMINMAX)
endif()

add_executable(zfilter-raw "${LIBRAW_SOURCE}/samples/dcraw_emu.cpp")
target_link_libraries(zfilter-raw PRIVATE zfilter_libraw)
